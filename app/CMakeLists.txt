cmake_minimum_required(VERSION 3.24)
project(VKomadoControl LANGUAGES CXX RC)
set(SHAREDVD_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../assets/sharedvd.ico")
configure_file(../assets/sharedvd.rc.in sharedvd.rc @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(PAYLOAD_DIR "" CACHE PATH "Verified driver payload")
foreach(name MttVDD.inf MttVDD.dll mttvdd.cat)
  file(SHA256 "${PAYLOAD_DIR}/${name}" hash)
  string(REPLACE "." "_" symbol "${name}")
  string(APPEND HASH_DEFINITIONS "inline constexpr wchar_t ${symbol}_hash[] = L\"${hash}\";\n")
endforeach()
configure_file(payload_hashes.h.in payload_hashes.h @ONLY)
add_library(sharedvd_core core.cpp)
target_include_directories(sharedvd_core PUBLIC "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(sharedvd_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(sharedvd_core PUBLIC setupapi cfgmgr32 newdev advapi32 bcrypt shell32 ole32 user32)
add_executable(VKomado.Device device_main.cpp "${CMAKE_CURRENT_BINARY_DIR}/sharedvd.rc")
target_link_libraries(VKomado.Device PRIVATE sharedvd_core)
add_executable(VKomado.Control WIN32 control.cpp "${CMAKE_CURRENT_BINARY_DIR}/sharedvd.rc")
target_link_libraries(VKomado.Control PRIVATE sharedvd_core comctl32)
add_executable(manager_tests ../tests/manager_tests.cpp)
target_link_libraries(manager_tests PRIVATE sharedvd_core)
add_executable(manager_ui_test ../tests/manager_ui_test.cpp)
target_link_libraries(manager_ui_test PRIVATE sharedvd_core gdiplus)
foreach(target sharedvd_core VKomado.Device VKomado.Control manager_tests manager_ui_test)
  target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
endforeach()
